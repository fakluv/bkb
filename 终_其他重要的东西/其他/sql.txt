        FROM dic_device_line_switch
    )
	
	
	
	报警查询具体到子系统：
	
	    SELECT
    t1.SYS_NO AS "子系统编号",
    t1.device_no AS "设备编号",
    t1.DEVICE_NAME AS "设备名称",

    t1.STATION_NO AS "站编号",
    t2.`LEVEL` AS "报警等级",
    t2.STATUS_NAME AS "设备状态",
    t4.`STATION_NAME`,
    CONCAT(t4.`STATION_NAME`, t1.DEVICE_NAME) AS first_device
    
FROM
    dic_device t1
    RIGHT JOIN dic_device_alarm t2 ON t1.`DEVICE_TYPE_NO` = t2.`DEVICE_TYPE_NO`
    LEFT JOIN dic_device_type t3 ON t1.`DEVICE_TYPE_NO` = t3.`DEVICE_TYPE_NO`
    LEFT JOIN dic_station t4 ON t4.`STATION_NO`=t1.`STATION_NO`
WHERE
    t2.`LEVEL` != 4
                AND t3.ALARM_FLAG = 1 
    AND t1.DEVICE_TYPE_NO NOT IN (
        SELECT DEVICE_TYPE_NO
        FROM dic_device_line_switch
    )
    AND t1.sys_no=1
    
								
包含所有报警等级报警条目的列表:								
                SELECT
        sys.SYS_NAME AS "系统名称",
        CASE 
        WHEN a.STATION_NO != -2 THEN st.STATION_NAME 
        ELSE CONCAT(
            SUBSTRING_INDEX(SUBSTRING_INDEX(a.device_no, '_', 3), '_', -1), 
            '车'
        )
    END AS "站点/车",
        a.device_no AS "故障码",
        alarm.`LEVEL` AS "报警等级",
        a.DEVICE_NAME AS "报警内容",
        alarm.STATUS_NAME AS "报警状态",
        a.DEVICE_TYPE_DESC AS "报警类型",
        ahelp.alarm_content AS "报警原因",
        ahelp.alarm_affect AS "报警影响",
        ahelp.suggestion AS "处置建议"
        
FROM
        dic_device a
        LEFT  JOIN dic_device_alarm alarm ON a.DEVICE_TYPE_NO = alarm.DEVICE_TYPE_NO
        LEFT JOIN dic_alarm_help ahelp ON alarm.alarm_id = ahelp.alarm_id
        LEFT JOIN dic_system sys ON a.SYS_NO = sys.SYS_NO
        LEFT JOIN dic_station st on a.STATION_NO = st.STATION_NO
WHERE  a.DEVICE_TYPE_NO IS NOT NULL AND a.device_no NOT IN (SELECT device_no FROM dic_device_line_switch);
